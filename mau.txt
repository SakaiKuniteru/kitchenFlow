import phongBanService from "./phong-ban.service.js";


class PhongBanController {
    async getDanhSach(req, res, next) {

        try {

            const data =
                await phongBanService.getDanhSach(
                    req.query
                );

            return res.status(200).json({

                success: true,

                message:
                    "Lấy danh sách phòng ban thành công.",

                data

            });

        } catch (error) {

            next(error);

        }

    }

    async getChiTiet(req, res, next) {

        try {

            const { id } = req.params;

            const data =
                await phongBanService.getChiTiet(id);

            return res.status(200).json({

                success: true,

                message:
                    "Lấy chi tiết phòng ban thành công.",

                data

            });

        } catch (error) {

            next(error);

        }

    }

    async create(req, res, next) {

        try {

            const data =
                await phongBanService.create(
                    req.body
                );

            return res.status(201).json({

                success: true,

                message:
                    "Thêm phòng ban thành công.",

                data

            });

        } catch (error) {

            next(error);

        }

    }


    async update(req, res, next) {

        try {

            const { id } = req.params;

            const data =
                await phongBanService.update(
                    id,
                    req.body
                );

            return res.status(200).json({

                success: true,

                message:
                    "Cập nhật phòng ban thành công.",

                data

            });

        } catch (error) {

            next(error);

        }

    }


    async khoa(req, res, next) {

        try {

            next(error);

        }

    }


    async moKhoa(req, res, next) {

        try {

            const { id } = req.params;

            await phongBanService.moKhoa(id);

            return res.status(200).json({

                success: true,

                message:
                    "Mở khóa phòng ban thành công.",

                data: null

            });

        } catch (error) {

            next(error);

        }

    }

}

.validate

const Joi = require("joi");

const updateSchema = Joi.object({

    maPhongBan: Joi.string()
        .trim()
        .max(50)
        .required()
        .messages({
            "string.empty":
                "Mã phòng ban không được để trống.",
            "string.max":
                "Mã phòng ban không được vượt quá 50 ký tự.",
            "any.required":
                "Mã phòng ban là bắt buộc."
        }),

    tenPhongBan: Joi.string()
        .trim()
        .max(255)
        .required()
        .messages({
            "string.empty":
                "Tên phòng ban không được để trống.",
            "string.max":
                "Tên phòng ban không được vượt quá 255 ký tự.",
            "any.required":
                "Tên phòng ban là bắt buộc."
        }),

    moTa: Joi.string()
        .trim()
        .max(500)
        .allow("", null),

    coSoId: Joi.number()
        .integer()
        .positive()
        .optional()
        .messages({
            "number.base":
                "ID cơ sở phải là số.",
            "number.integer":
                "ID cơ sở không hợp lệ.",
            "number.positive":
                "ID cơ sở không hợp lệ."
        }),

    maCoSo: Joi.string()
        .trim()
        .max(50)
        .optional()
        .messages({
            "string.empty":
                "Mã cơ sở không được để trống.",
            "string.max":
                "Mã cơ sở không được vượt quá 50 ký tự."
        })

})
    .or(
        "coSoId",
        "maCoSo"
    )
    .messages({

        "object.missing":
            "Phải truyền coSoId hoặc maCoSo."

    });

module.exports = {
    updateSchema
};
